Usage

Load Moment.js first, then a Moment Timezone build that includes zone data.

<script src="assets/extensions/moment/min/moment.min.js"></script>
<script src="assets/extensions/moment-timezone/builds/moment-timezone-with-data-10-year-range.min.js"></script>

moment.tz.guess()                                   // "Asia/Jakarta"
moment().tz('Asia/Tokyo').format('YYYY-MM-DD HH:mm z')
moment.tz('2026-10-21 09:00', 'Asia/Jakarta').tz('America/New_York').format()
